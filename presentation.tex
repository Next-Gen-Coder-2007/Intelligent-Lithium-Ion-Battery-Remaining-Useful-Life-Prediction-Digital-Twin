\begin{frame}{Problem Statement: Data-Driven RUL Prediction}
\begin{columns}[T]
  \begin{column}{0.52\textwidth}
    \textbf{Operational Degradation Challenges in Dataset:}
    \begin{itemize}\footnotesize
      \item \textbf{Non-Invasive Prognostics Need}: Internal battery capacity cannot be measured directly during online operation; RUL must be estimated solely from surface operational telemetry.
      \item \textbf{Non-Linear Feature Dynamics}: Operational features evolve non-linearly over 1,133 lifetime cycles ($r = +0.7891$ for discharge cutoff voltage; $r = -0.7682$ for charging onset voltage).
      \item \textbf{High Multicollinearity}: Charging duration features exhibit strong collinearity ($\text{VIF} > 10$), causing standard linear regression models to fail ($R^2 = 0.7762$).
    \end{itemize}
  \end{column}

  \begin{column}{0.44\textwidth}
    \begin{block}{\textbf{Mathematical Formulation}}
      \footnotesize
      Given 7 standardized operational features $\mathbf{x} \in \mathbb{R}^7$, learn a mapping function $f: \mathbb{R}^7 \to \mathbb{R}$ to predict remaining cycles:
      $$\hat{y} = f(\mathbf{x}) \approx \text{RUL} \in [0, 1133]$$
      \textbf{Optimization Objective:}
      $$\min_{\theta} \frac{1}{N}\sum_{i=1}^{N} |y_i - f(\mathbf{x}_i; \theta)| \quad (\text{MAE})$$
    \end{block}
    \vspace{0.2em}
    \begin{alertblock}{\textbf{Engineering Target}}
      \scriptsize Build a leak-free, tuned ML pipeline achieving test $\text{MAE} < 10$ cycles across the complete battery lifecycle.
    \end{alertblock}
  \end{column}
\end{columns}
\end{frame}
